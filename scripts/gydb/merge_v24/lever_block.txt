
## ===== MERGE_V24 (pass 24): v12.11 levers 2 and 4 re-added on top of the v13.2 stack. =====
## AcadianGY_13.1.r already implements both gated ops switches (mort.dens.scale line ~4137,
## use.dht.hw.mod line ~1847); this block only sets ops$, exactly as the deleted V1211_* block
## of bench_parity_split_ms_v1211.R did. Defaults FALSE, so with neither set this file is
## behaviourally identical to p21/2026-08-19_p21_bench_v132.R (the v13.2 harness).
V1211_MORTGATE <- as.logical(Sys.getenv("V1211_MORTGATE","FALSE"))
V1211_MANAGED  <- as.logical(Sys.getenv("V1211_MANAGED","FALSE"))
V1211_DHTHW    <- as.logical(Sys.getenv("V1211_DHTHW","FALSE"))
if (isTRUE(V1211_MORTGATE)) { ops$mort.dens.scale <- TRUE; ops$stand.managed <- isTRUE(V1211_MANAGED)
  ops$mort.dens.fmin <- as.numeric(Sys.getenv("MORT_FMIN","0.13"))
  ops$mort.dens.k    <- as.numeric(Sys.getenv("MORT_K","12.4"))
  ops$mort.dens.rd0  <- as.numeric(Sys.getenv("MORT_RD0","0.38")) }
if (isTRUE(V1211_DHTHW))    { ops$use.dht.hw.mod <- TRUE }
cat(sprintf("MERGE_V24 levers: MORTGATE=%s(managed=%s) DHTHW=%s\n", V1211_MORTGATE, V1211_MANAGED, V1211_DHTHW))
